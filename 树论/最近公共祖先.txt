// dfs求最近公共祖先
#include <bits/stdc++.h>
using namespace std;

#define int long long

const int MAXN = 2e5 + 5;
const int LOG = 20;

vector<vector<int>> tree(MAXN);
vector<int> depth(MAXN);

// fa[x][j]：x 向上跳 2^j 步到达的祖先
int fa[MAXN][LOG + 1];

// 预处理深度和倍增祖先
void dfs(int x, int father)
{
    // x 的直接父亲
    fa[x][0] = father;

    // 计算 x 向上跳 2^i 步的祖先
    for (int i = 1; i <= LOG; i++)
    {
        fa[x][i] = fa[fa[x][i - 1]][i - 1];
    }

    for (int i = 0; i < tree[x].size(); i++)
    {
        int now = tree[x][i];

        // 无向边，不能走回父亲
        if (now == father)
        {
            continue;
        }

        depth[now] = depth[x] + 1;
        dfs(now, x);
    }
}

// 求 x 和 y 的最近公共祖先
int lca(int x, int y)
{
    // 保证 x 是深度更大的节点
    if (depth[x] < depth[y])
    {
        swap(x, y);
    }

    // 第一步：让 x 和 y 跳到相同深度
    for (int i = LOG; i >= 0; i--)
    {
        if (depth[fa[x][i]] >= depth[y])
        {
            x = fa[x][i];
        }
    }

    // 如果此时相等，说明 y 原本就是 x 的祖先
    if (x == y)
    {
        return x;
    }

    // 第二步：x 和 y 一起向上跳
    // 保证跳完以后它们仍然不相等
    for (int i = LOG; i >= 0; i--)
    {
        if (fa[x][i] != fa[y][i])
        {
            x = fa[x][i];
            y = fa[y][i];
        }
    }

    // 此时 x 和 y 的父亲就是最近公共祖先
    return fa[x][0];
}

signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    int n;
    cin >> n;

    for (int i = 1; i < n; i++)
    {
        int x, y;
        cin >> x >> y;

        // 无向树，两边都要存
        tree[x].push_back(y);
        tree[y].push_back(x);
    }

    // 规定 1 号节点为根
    // 0 号节点作为不存在的祖先
    depth[0] = 0;
    depth[1] = 1;

    dfs(1, 0);

    int q;
    cin >> q;

    while (q--)
    {
        int x, y;
        cin >> x >> y;

        cout << lca(x, y) << '\n';
    }

    return 0;
}

// 栈模拟dfs求欧拉序列求最近公共祖先
#include <bits/stdc++.h>
using namespace std;

/*
DFS 欧拉序 + ST 表求 LCA

特点：

1. 使用栈模拟 DFS，不会因为树是一条链而爆栈
2. 预处理复杂度 O(n log n)
3. 单次查询复杂度 O(1)

使用方法：

buildLCA(1);       // 以 1 为根预处理
int z = LCA(x, y); // 查询 x、y 的最近公共祖先

注意：

这里不使用 #define int long long。
因为节点编号和深度使用普通 int 足够，
而 ST 表很大，使用 long long 会让内存翻倍。
*/

const int MAXN = 5e5 + 5;
const int MAXM = MAXN * 2;
const int LOG = 21;

vector<int> tree[MAXN];

int depth[MAXN];

// first[x]：x 在欧拉序中第一次出现的位置
int first[MAXN];

// euler[i]：欧拉序第 i 个位置的节点
int euler[MAXM];

int lg[MAXM];

// st[k][i]：从 i 开始，长度为 2^k 的区间中深度最小的节点
int st[LOG + 1][MAXM];

int cnt;

// 使用栈模拟 DFS，建立欧拉序
void dfs(int root)
{
    typedef struct Node
    {
        int x;
        int father;

        // 下一个准备访问的邻接点下标
        int pos;
    } Node;

    vector<Node> stk;

    depth[root] = 0;

    first[root] = ++cnt;
    euler[cnt] = root;

    stk.push_back({root, 0, 0});

    while (!stk.empty())
    {
        Node &top = stk.back();

        // 当前节点的所有邻接点都处理完了
        if (top.pos == tree[top.x].size())
        {
            stk.pop_back();

            // 从儿子返回父亲，再记录一次父亲
            if (!stk.empty())
            {
                euler[++cnt] = stk.back().x;
            }

            continue;
        }

        int now = tree[top.x][top.pos];
        top.pos++;

        if (now == top.father)
        {
            continue;
        }

        depth[now] = depth[top.x] + 1;

        // 第一次进入 now
        first[now] = ++cnt;
        euler[cnt] = now;

        stk.push_back({now, top.x, 0});
    }
}

// 返回深度更小的节点
int getMin(int x, int y)
{
    if (depth[x] < depth[y])
    {
        return x;
    }

    return y;
}

void buildLCA(int root)
{
    cnt = 0;

    // 建立欧拉序
    dfs(root);

    // 预处理 floor(log2(i))
    lg[1] = 0;

    for (int i = 2; i <= cnt; i++)
    {
        lg[i] = lg[i / 2] + 1;
    }

    // ST 表初始状态
    for (int i = 1; i <= cnt; i++)
    {
        st[0][i] = euler[i];
    }

    // 建立 ST 表
    for (int k = 1; (1 << k) <= cnt; k++)
    {
        for (int i = 1; i + (1 << k) - 1 <= cnt; i++)
        {
            int left_node = st[k - 1][i];

            int right_node =
                st[k - 1][i + (1 << (k - 1))];

            st[k][i] = getMin(left_node, right_node);
        }
    }
}

int LCA(int x, int y)
{
    int l = first[x];
    int r = first[y];

    if (l > r)
    {
        swap(l, r);
    }

    int len = r - l + 1;
    int k = lg[len];

    int left_node = st[k][l];
    int right_node = st[k][r - (1 << k) + 1];

    return getMin(left_node, right_node);
}

signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);

    int n, q;
    cin >> n >> q;

    for (int i = 1; i < n; i++)
    {
        int u, v;
        cin >> u >> v;

        tree[u].push_back(v);
        tree[v].push_back(u);
    }

    buildLCA(1);

    while (q--)
    {
        int x, y;
        cin >> x >> y;

        cout << LCA(x, y) << '\n';
    }

    return 0;
}
