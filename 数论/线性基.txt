线性基（XOR）通用构造模板
====================

功能：
- 插入数字
- 判断 x 能否被某个子集 XOR 得到
- 返回一组构造出 x 的原始编号
- 最大子集 XOR
- 判断是否存在非空子集 XOR = 0

默认：
- 原数组编号从 1 开始
- long long
- 数据非负，位数 0 ~ 62
- MAXN 改成题目 n 的上限


#include <bits/stdc++.h>
using namespace std;

#define int long long

const int LOG = 62;
const int MAXN = 2005;

struct LinearBasis
{
    // basis[i]：最高位为 i 的基底
    int basis[LOG + 1];

    // from[i][j] = 1：
    // 第 j 个原始数参与构造 basis[i]
    bitset<MAXN> from[LOG + 1];

    int rk;
    int have_zero;

    LinearBasis()
    {
        clear();
    }

    void clear()
    {
        for (int i = 0; i <= LOG; i++)
        {
            basis[i] = 0;
            from[i].reset();
        }

        rk = 0;
        have_zero = 0;
    }

    // 插入第 id 个原始数 x
    //
    // 返回：
    // 1：插入成功，成为新基底
    // 0：线性相关
    int insert(int x, int id)
    {
        bitset<MAXN> cur;
        cur[id] = 1;

        for (int i = LOG; i >= 0; i--)
        {
            if (((x >> i) & 1) == 0)
            {
                continue;
            }

            if (basis[i] == 0)
            {
                basis[i] = x;
                from[i] = cur;
                rk++;

                return 1;
            }

            x ^= basis[i];
            cur ^= from[i];
        }

        have_zero = 1;

        return 0;
    }

    // 判断 x 能不能被已有数字 XOR 出来
    int check(int x)
    {
        for (int i = LOG; i >= 0; i--)
        {
            if (((x >> i) & 1) == 0)
            {
                continue;
            }

            if (basis[i] == 0)
            {
                return 0;
            }

            x ^= basis[i];
        }

        return 1;
    }

    // 返回一组构造出 x 的原始编号
    //
    // 返回：
    // first = 0：无法构造
    // first = 1：可以构造
    //
    // second 中所有为 1 的位置：
    // 就是参与 XOR 出 x 的原始编号
    pair<int, bitset<MAXN>> get(int x)
    {
        bitset<MAXN> cur;

        for (int i = LOG; i >= 0; i--)
        {
            if (((x >> i) & 1) == 0)
            {
                continue;
            }

            if (basis[i] == 0)
            {
                return {0, cur};
            }

            x ^= basis[i];
            cur ^= from[i];
        }

        return {1, cur};
    }

    // 最大化：
    // x XOR 某个子集 XOR
    //
    // 默认 x = 0：
    // 最大子集 XOR
    int query_max(int x = 0)
    {
        for (int i = LOG; i >= 0; i--)
        {
            x = max(x, x ^ basis[i]);
        }

        return x;
    }
};


--------------------------------------------------
1. 插入
--------------------------------------------------

LinearBasis lb;

for (int i = 1; i <= n; i++)
{
    int x;
    cin >> x;

    lb.insert(x, i);
}


--------------------------------------------------
2. 判断 x 能不能构造
--------------------------------------------------

if (lb.check(x))
{
    cout << "YES\n";
}
else
{
    cout << "NO\n";
}


--------------------------------------------------
3. 返回构造出 x 的全部原始编号
--------------------------------------------------

auto [flag, res] = lb.get(x);

if (flag == 0)
{
    cout << "NO\n";
}
else
{
    cout << "YES\n";

    for (int i = 1; i <= n; i++)
    {
        if (res[i])
        {
            cout << i << " ";
        }
    }

    cout << '\n';
}


--------------------------------------------------
4. 返回构造出 x 的原始值
--------------------------------------------------

假设原数组是：

vector<int> a(n + 1);

得到 res 后：

for (int i = 1; i <= n; i++)
{
    if (res[i])
    {
        cout << a[i] << " ";
    }
}


--------------------------------------------------
5. 最大子集 XOR
--------------------------------------------------

cout << lb.query_max() << '\n';


--------------------------------------------------
6. 给初始值 x，最大化 x XOR subset
--------------------------------------------------

cout << lb.query_max(x) << '\n';


--------------------------------------------------
7. 是否存在非空子集 XOR = 0
--------------------------------------------------

if (lb.have_zero)
{
    cout << "YES\n";
}

或者：

if (lb.rk < n)
{
    cout << "YES\n";
}


--------------------------------------------------
8. get(0)
--------------------------------------------------

get(0) 默认返回：

    可以构造
    构造集合为空

因为空集 XOR = 0。

如果题目要求：

    非空子集 XOR = 0

不要用 get(0)。

直接判断：

    have_zero

如果还需要返回一组非空 XOR = 0 的方案，
则需要在 insert 时额外保存线性相关时得到的 cur。


--------------------------------------------------
9. 赛场速记
--------------------------------------------------

插入：

    lb.insert(x, id);


判断能否构造：

    lb.check(x);


返回构造：

    auto [flag, res] = lb.get(x);


最大 XOR：

    lb.query_max();


来源同步：

    x ^= basis[i];
    cur ^= from[i];


一句话：

    basis 存值
    from 存这个值由哪些原始数构造
    get(x) 返回一组 XOR 等于 x 的原始编号
