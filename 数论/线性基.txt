线性基（XOR）通用赛场板子
====================

功能：
- 插入数字
- 判断某个值能不能被子集 XOR 出来
- 返回一组构造出 x 的原始编号
- 最大子集 XOR
- 最小化 x XOR subset
- 判断是否存在非空子集 XOR = 0
- 第 k 小不同 XOR
- 合并两个线性基
- 统计不同 XOR 结果个数
- 统计 XOR = x 的子集数量

默认：
- 原数组编号从 1 开始
- long long
- 数据非负，位数 0 ~ 62
- MAXN 改成题目 n 上限以上


#include <bits/stdc++.h>
using namespace std;

#define int long long

const int LOG = 62;
const int MAXN = 2005;

struct LinearBasis
{
    // basis[i]：最高位为 i 的基底
    int basis[LOG + 1];

    // from[i][j] = 1：
    // 第 j 个原始数参与构造 basis[i]
    bitset<MAXN> from[LOG + 1];

    int rk;
    int have_zero;

    LinearBasis()
    {
        clear();
    }

    void clear()
    {
        for (int i = 0; i <= LOG; i++)
        {
            basis[i] = 0;
            from[i].reset();
        }

        rk = 0;
        have_zero = 0;
    }

    // 插入第 id 个原始数 x
    //
    // 返回：
    // 1：插入成功，成为新基底
    // 0：线性相关
    int insert(int x, int id)
    {
        bitset<MAXN> cur;
        cur[id] = 1;

        for (int i = LOG; i >= 0; i--)
        {
            if (((x >> i) & 1) == 0)
            {
                continue;
            }

            if (basis[i] == 0)
            {
                basis[i] = x;
                from[i] = cur;
                rk++;

                return 1;
            }

            x ^= basis[i];
            cur ^= from[i];
        }

        have_zero = 1;

        return 0;
    }

    // 判断 x 是否能由某个子集 XOR 得到
    int check(int x)
    {
        for (int i = LOG; i >= 0; i--)
        {
            if (((x >> i) & 1) == 0)
            {
                continue;
            }

            if (basis[i] == 0)
            {
                return 0;
            }

            x ^= basis[i];
        }

        return 1;
    }

    // 返回一组构造出 x 的原始编号
    //
    // first = 0：无法构造
    // first = 1：可以构造
    //
    // second 中为 1 的位置：
    // 就是参与 XOR 出 x 的原始编号
    pair<int, bitset<MAXN>> get(int x)
    {
        bitset<MAXN> cur;

        for (int i = LOG; i >= 0; i--)
        {
            if (((x >> i) & 1) == 0)
            {
                continue;
            }

            if (basis[i] == 0)
            {
                return {0, cur};
            }

            x ^= basis[i];
            cur ^= from[i];
        }

        return {1, cur};
    }

    // 最大化：
    // x XOR 某个子集 XOR
    //
    // 默认 x = 0：
    // 最大子集 XOR
    int query_max(int x = 0)
    {
        for (int i = LOG; i >= 0; i--)
        {
            x = max(x, x ^ basis[i]);
        }

        return x;
    }

    // 最小化：
    // x XOR 某个子集 XOR
    int query_min(int x)
    {
        for (int i = LOG; i >= 0; i--)
        {
            x = min(x, x ^ basis[i]);
        }

        return x;
    }

    // 化成适合求第 k 小的形式
    // 使用 kth 前先调用
    void rebuild()
    {
        for (int i = 0; i <= LOG; i++)
        {
            if (basis[i] == 0)
            {
                continue;
            }

            for (int j = i + 1; j <= LOG; j++)
            {
                if ((basis[j] >> i) & 1)
                {
                    basis[j] ^= basis[i];
                    from[j] ^= from[i];
                }
            }
        }
    }

    // 第 k 小“不同 XOR 值”
    // k 从 0 开始
    //
    // kth(0) = 0
    //
    // 使用前：
    // rebuild()
    //
    // 不存在返回 -1
    int kth(int k)
    {
        if (rk < 63 && k >= (1LL << rk))
        {
            return -1;
        }

        vector<int> temp;

        for (int i = 0; i <= LOG; i++)
        {
            if (basis[i])
            {
                temp.push_back(basis[i]);
            }
        }

        int ans = 0;

        for (int i = 0; i < temp.size(); i++)
        {
            if ((k >> i) & 1)
            {
                ans ^= temp[i];
            }
        }

        return ans;
    }

    // 合并另一个线性基
    //
    // 只用于值域性质：
    // query_max / check / kth 等
    //
    // 不建议依赖 merge 后的 from 做原编号构造
    void merge(const LinearBasis &other)
    {
        for (int i = 0; i <= LOG; i++)
        {
            if (other.basis[i])
            {
                int x = other.basis[i];

                for (int j = LOG; j >= 0; j--)
                {
                    if (((x >> j) & 1) == 0)
                    {
                        continue;
                    }

                    if (basis[j] == 0)
                    {
                        basis[j] = x;
                        rk++;
                        break;
                    }

                    x ^= basis[j];
                }

                if (x == 0)
                {
                    have_zero = 1;
                }
            }
        }

        if (other.have_zero)
        {
            have_zero = 1;
        }
    }
};

1. 插入
--------------------------------------------------

LinearBasis lb;

for (int i = 1; i <= n; i++)
{
    int x;
    cin >> x;

    lb.insert(x, i);
}

cout << lb.query_max() << '\n';


--------------------------------------------------

4. 最小化 x XOR subset
--------------------------------------------------

--------------------------------------------------
5. 判断 x 能不能被 XOR 出来

if (lb.check(x))

    cout << "YES\n";
}

{
    cout << "NO\n";
}

6. 返回一组构造出 x 的原始编号

auto [flag, res] = lb.get(x);

if (flag == 0)
{
    cout << "NO\n";
}
else
{
    cout << "YES\n";

    for (int i = 1; i <= n; i++)
    {
        if (res[i])
        {
            cout << i << " ";
        }
    }

    cout << '\n';
}


--------------------------------------------------
7. 返回构造出 x 的原始值

假设：

vector<int> a(n + 1);

得到 res 后：

for (int i = 1; i <= n; i++)

    if (res[i])
    {
        cout << a[i] << " ";
    }
}


--------------------------------------------------
8. rank
--------------------------------------------------

lb.rk

表示线性无关基底数量。

不同 XOR 结果个数：

    2^rk


--------------------------------------------------
9. XOR = x 的子集数量
--------------------------------------------------

如果：

    lb.check(x) == 0

答案：

    0


如果：

    lb.check(x) == 1

原数组一共有 n 个数：

    XOR = x 的子集数量 = 2^(n - rk)


--------------------------------------------------
10. 是否存在非空子集 XOR = 0
--------------------------------------------------

方法 1：

if (lb.have_zero)
{
    ...
}


方法 2：

if (lb.rk < n)
{
    ...
}


--------------------------------------------------
11. 第 k 小不同 XOR
--------------------------------------------------

先：

lb.rebuild();


然后：

lb.kth(k);


k 从 0 开始：

    kth(0) = 0


如果题目说：

    第 k 小不同 XOR
    k 从 1 开始
    0 算进去

调用：

    lb.kth(k - 1);


--------------------------------------------------
12. 第 k 小非空子集 XOR
--------------------------------------------------

如果：

    lb.have_zero == 1

说明非空子集也可以得到 0：

    ans = lb.kth(k - 1);


如果：

    lb.have_zero == 0

说明 0 只能由空集得到：

    ans = lb.kth(k);


--------------------------------------------------
13. 最小非零子集 XOR
--------------------------------------------------

先：

lb.rebuild();


然后：

int ans = 0;

for (int i = 0; i <= LOG; i++)
{
    if (lb.basis[i])
    {
        ans = lb.basis[i];
        break;
    }
}


--------------------------------------------------
14. 合并两个线性基
--------------------------------------------------

LinearBasis A;
LinearBasis B;

...

A.merge(B);


注意：

    merge 主要用于值域性质。

    如果你还要 get(x) 返回原始编号，
    最好所有原始数字直接插入同一个 LinearBasis，
    不要依赖 merge 后的 from。


--------------------------------------------------
15. get(0)
--------------------------------------------------

get(0) 返回：

    first = 1
    second 为空

表示：

    空集 XOR = 0


如果题目要求：

    非空子集 XOR = 0

直接判断：

    have_zero

如果还需要输出一组非空 XOR = 0 的方案，
需要在 insert 线性相关时把当时的 cur 保存下来。


--------------------------------------------------
16. 赛场速记
--------------------------------------------------

插入：

    lb.insert(x, id);


判断能否构造：

    lb.check(x);


返回构造：

    auto [flag, res] = lb.get(x);


最大 XOR：

    lb.query_max();


最小化：

    lb.query_min(x);


第 k 小：

    lb.rebuild();
    lb.kth(k);


合并：

    A.merge(B);


非空 XOR = 0：

    lb.have_zero
    或
    lb.rk < n


来源同步：

    x ^= basis[i];
    cur ^= from[i];


一句话：

    basis 存基底值
    from 存基底由哪些原始编号构成
    get(x) 返回一组 XOR 等于 x 的原始编号
