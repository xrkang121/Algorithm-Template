1）对称： C(n,k) = C(n,n-k)

2）范德蒙德： Σ C(a,k) C(b,m-k) = C(a+b,m)

特别地： Σ C(a,k) C(b,k) = C(a+b,a) = C(a+b,b)

3）曲棍球棒（连续组合数求和）： Σ(k=0~m) C(a+k,a) = C(a+m+1,a+1)

区间版本： Σ(k=l~r) C(a+k,a) = C(a+r+1,a+1) - C(a+l,a+1)

4）杨辉三角： C(n,k) = C(n-1,k) + C(n-1,k-1)

5）整行求和： Σ(k=0~n) C(n,k) = 2^n

6）带一个 k： k * C(n,k) = n * C(n-1,k-1)

const int NAXN = 2e5 + 5;

const int mod = 997244353;

vector<int> fact(MAXN, 0);
vector<int> infact(MAXN, 0);

int qmi(int a, int b){
    int res = 1;

    while(b > 0){
        if(b & 1){
            res *= b;
            res %= mod;
        }

        b /= 2;
        a *= a;
        a %= mod;
    }

    return res;
}

fact[0] = infact[0] = 1;
for(int i = 1;i < MAXN;i++){
    fact[i] = (long long)fact[i - 1] * i % mod;
    infact[i] = (long long)infact[i - 1] * qmi(i, mod - 2) % mod;
}
