【基础结构】

typedef struct Point
{
    int x, y;
} Point;

int cross(Point a, Point b, Point c)
{
    return (b.x - a.x) * (c.y - a.y)
         - (b.y - a.y) * (c.x - a.x);
}

long double dis(Point a, Point b)
{
    long double dx = a.x - b.x;
    long double dy = a.y - b.y;

    return sqrtl(dx * dx + dy * dy);
}

int dist2(Point a, Point b)
{
    int dx = a.x - b.x;
    int dy = a.y - b.y;

    return dx * dx + dy * dy;
}


【叉积】

cross(A,B,C) > 0：左转
cross(A,B,C) < 0：右转
cross(A,B,C) = 0：共线


【完整凸包】

// 返回逆时针凸包
// 共线边只保留最外侧端点
vector<Point> convexHull(vector<Point> arr)
{
    sort(arr.begin(), arr.end(), [](Point a, Point b)
    {
        if (a.x != b.x)
        {
            return a.x < b.x;
        }

        return a.y < b.y;
    });

    arr.erase(unique(arr.begin(), arr.end(), [](Point a, Point b)
    {
        return a.x == b.x && a.y == b.y;
    }), arr.end());

    int n = arr.size();

    if (n <= 2)
    {
        return arr;
    }

    vector<Point> hull;

    for (int i = 0; i < n; i++)
    {
        while (hull.size() >= 2 &&
               cross(hull[hull.size() - 2], hull.back(), arr[i]) <= 0)
        {
            hull.pop_back();
        }

        hull.push_back(arr[i]);
    }

    int siz = hull.size();

    for (int i = n - 2; i >= 0; i--)
    {
        while ((int)hull.size() > siz &&
               cross(hull[hull.size() - 2], hull.back(), arr[i]) <= 0)
        {
            hull.pop_back();
        }

        hull.push_back(arr[i]);
    }

    hull.pop_back();

    return hull;
}


【下凸包】

// x 从小到大
// 保留左转
// cross <= 0 弹
vector<Point> lowerHull(vector<Point> arr)
{
    sort(arr.begin(), arr.end(), [](Point a, Point b)
    {
        if (a.x != b.x)
        {
            return a.x < b.x;
        }

        return a.y < b.y;
    });

    vector<Point> hull;

    for (int i = 0; i < arr.size(); i++)
    {
        while (hull.size() >= 2 &&
               cross(hull[hull.size() - 2], hull.back(), arr[i]) <= 0)
        {
            hull.pop_back();
        }

        hull.push_back(arr[i]);
    }

    return hull;
}


【上凸包】

// x 从小到大
// 保留右转
// cross >= 0 弹
vector<Point> upperHull(vector<Point> arr)
{
    sort(arr.begin(), arr.end(), [](Point a, Point b)
    {
        if (a.x != b.x)
        {
            return a.x < b.x;
        }

        return a.y < b.y;
    });

    vector<Point> hull;

    for (int i = 0; i < arr.size(); i++)
    {
        while (hull.size() >= 2 &&
               cross(hull[hull.size() - 2], hull.back(), arr[i]) >= 0)
        {
            hull.pop_back();
        }

        hull.push_back(arr[i]);
    }

    return hull;
}


【是否保留共线点】

完整凸包 / 下凸包：
<= 0  -> 不保留共线中间点
< 0   -> 保留共线点

上凸包：
>= 0  -> 不保留共线中间点
> 0   -> 保留共线点


【周长】

L = Σ dis(P[i], P[(i+1)%m])

long double perimeter(vector<Point> hull)
{
    long double ans = 0;
    int m = hull.size();

    for (int i = 0; i < m; i++)
    {
        ans += dis(hull[i], hull[(i + 1) % m]);
    }

    return ans;
}


【面积】

2S = |Σ(x[i]*y[i+1] - y[i]*x[i+1])|

int area2(vector<Point> hull)
{
    int ans = 0;
    int m = hull.size();

    for (int i = 0; i < m; i++)
    {
        Point a = hull[i];
        Point b = hull[(i + 1) % m];

        ans += a.x * b.y - a.y * b.x;
    }

    return llabs(ans);
}


【重要性质】

1. 最远点对一定在凸包点上
2. 最大/最小线性函数一定在凸包上
3. 给定方向 v：
   max(P·v)、min(P·v) 一定在凸包边界
4. 很多极值题：
   原点集 -> 求凸包 -> 只处理凸包点
5. 凸包复杂度 O(n log n)


【直线转点】

直线：

y = kx + b

转成点：

P = (k,b)

固定 x=t：

y = kt+b
  = (k,b)·(t,1)

结论：

求某条直线能否成为最高线
<=> 求点 (k,b) 是否可能使点积最大
<=> 看上凸包

求某条直线能否成为最低线
<=> 看下凸包


【上包络 / 下包络】

上包络：

相同 k：
只保留最大 b

然后：
求 (k,b) 的上凸包


下包络：

相同 k：
只保留最小 b

然后：
求 (k,b) 的下凸包


【两直线交点与点间斜率】

两直线：

y = k1*x + b1
y = k2*x + b2

交点横坐标：

x = (b2-b1)/(k1-k2)

转换点：

P1=(k1,b1)
P2=(k2,b2)

点间斜率：

slope(P1,P2)
=
(b2-b1)/(k2-k1)

所以：

交点横坐标
=
-slope(P1,P2)


【常见凸包转换】

1. 直线上包络
   y=kx+b -> 点(k,b) -> 上凸包

2. 直线下包络
   y=kx+b -> 点(k,b) -> 下凸包

3. 最大化：
   a*x+b*y
   = P·(a,b)
   -> 极值只看凸包

4. DP 斜率优化：
   dp[i] = min/max(m[j]*x[i]+b[j])
   -> 维护直线包络
   -> 本质凸包

5. 最远点对
   -> 凸包 + 旋转卡壳

6. 最大三角形面积
   -> 先求凸包

7. 最小面积矩形
   -> 凸包 + 旋转卡壳
